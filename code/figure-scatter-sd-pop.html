library(tidyverse)
library(MARSS)
library(here)
library(ggplot2)
library(ggExtra)
library(gridExtra)
res <- readRDS(here("results", "adjusted_model_fits.rds"))
ratio_unpooled_list <- res$ratio_unpooled_list
sigma.vec <- res$sigma.vec
my_states <- res$my_states

pop_89 <- read.csv(here("data", "89_pop.csv"), header = TRUE, row.names = 1)
pop_89 <- pop_89["Jul.89"]
# sd_ratio = 3, so persistent variance = 9 * q (transitory variance)
sd_ratio <- 3

q_t <- sapply(ratio_unpooled_list, function(x) coef(x)$Q[1, 1])
drift <- sapply(ratio_unpooled_list, function(x) coef(x)$U[1, 1])

q_p <- sd_ratio^2 * q_t
sampling_var <- sigma.vec[my_states]^2

state_bfgs_table <- data.frame(
  drift          = drift,
  process_SD     = sqrt(q_p),                       # persistent component
  shock_SD       = sqrt(q_t),                       # transitory component
  sampling_SD    = sqrt(sampling_var),
  observation_SD = sqrt(q_t + sampling_var),        # total obs noise
  row.names      = my_states
)
pop_89_for_figure = subset(pop_89, !(rownames(pop_89) %in% c("US", "DC")))
pop_89_for_figure = pop_89_for_figure / 1e6

shared_states <- intersect(rownames(state_bfgs_table), rownames(pop_89_for_figure))
state_table_for_figure <- cbind(
  state_bfgs_table[shared_states, ],
  pop_89_for_figure[shared_states, , drop = FALSE]
)
state_table_for_figure = as.data.frame(state_table_for_figure)
state_table_for_figure[] = lapply(state_table_for_figure, function(x) as.numeric(x))
# remove states with 0 observation variance
state_table_for_figure = subset(state_table_for_figure, state_table_for_figure$observation_SD >= 0.000000001)
p1 <- ggplot(state_table_for_figure, aes(x = Jul.89, y = process_SD)) +
  geom_point() +
  geom_smooth(method = "lm", se = FALSE, col = "red") +
  theme_minimal() +
  ylim(0, 0.5) +
  labs(x = "Population Size (In Millions)", y = "Standard Deviation in Years", title = "(a) Process Standard Deviation")
